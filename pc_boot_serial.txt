Serial Port, boot sector program

Read data from serial port and print on screen:

[bits 16]
[org 0x7c00]
port equ 0x3f8

  mov ah, 0x00      ; Init port
  mov al, 0xe3      ; Config (11100011b = 9600, n, 8, 1)
  mov dx, 0x00      ; Com1
  int 0x14

read_serial:
  mov dx, port + 5  ; Line Status Register
.wait:
  in  al, dx        ; Read status
  test al, 1        ; Has bit 0 (Data Ready) been set?
  jz  .wait         ; No, wait
  mov dx, port         
  in  al, dx        ; Read byte from Receive Buffer into AL
  mov ah, 0x0e
  int 0x10
  jmp read_serial

times 510-($-$$) db 0
dw 0xaa55
