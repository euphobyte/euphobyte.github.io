C# Serial Port

Source for a dumpserial command:

using System;
using System.IO.Ports;
using System.Threading;

public class PortChat{

  public static void Main(){
    SerialPort sp = new SerialPort("COM3", 9600);
    sp.Open();
    int i = 0;
    string str="";
    while(true){
      if(sp.BytesToRead > 0){
        byte b = (byte)sp.ReadByte();
        Console.Write("{0:X2} ", b);
        if(b > 32)
          str += (char) b;
        else
          str += ".";
        if(i==15){
          Console.WriteLine("  " + str);
          str = "";
          i = 0;
        }
        i++;
      }
    }
  }
}

Save this program as dumpserial.cs then compile:

> csc dumpserial.cs
